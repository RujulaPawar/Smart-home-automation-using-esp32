#include "DHT.h"
#include <ESP32Servo.h>
#include <Wire.h>
#include <LiquidCrystal_PCF8574.h>

// Pin Definitions
#define MQ2_PIN 34
#define GAS_THRESHOLD 3000

#define DHT_PIN 26
#define DHT_TYPE DHT11
DHT dht(DHT_PIN, DHT_TYPE);

#define FLAME_SENSOR_PIN 13
#define LDR_PIN 35
#define LIGHT_THRESHOLD 3500

#define PIR_PIN 23
#define RAINDROP_PIN 32
#define RAIN_THRESHOLD 2800

// Output Pins
#define LED_RED 2
#define LED_GREEN 4
#define BUZZER 5
#define IR_SENSOR_PIN 14

// Servo Pins
#define SERVO1_PIN 12  // Gas leak servo
#define SERVO2_PIN 27  // Flame servo
#define SERVO3_PIN 25  // Rain servo

Servo servo1;
Servo servo2;
Servo servo3;

LiquidCrystal_PCF8574 lcd(0x27);  // Try 0x3F if this doesn't work

bool alertActive = false;
bool gasAlert = false;
bool flameAlert = false;
bool motionAlert = false;
bool rainAlert = false;

void setup() {
  Serial.begin(115200);
  
  // Initialize sensors
  pinMode(MQ2_PIN, INPUT);
  pinMode(FLAME_SENSOR_PIN, INPUT);
  pinMode(LDR_PIN, INPUT);
  pinMode(PIR_PIN, INPUT);
  pinMode(RAINDROP_PIN, INPUT);
  pinMode(IR_SENSOR_PIN, INPUT);
  dht.begin();
  
  // Initialize outputs
  pinMode(LED_RED, OUTPUT);
  pinMode(LED_GREEN, OUTPUT);
  pinMode(BUZZER, OUTPUT);
  pinMode(18, OUTPUT);
  pinMode(19, OUTPUT);
  
  // Attach servos
  servo1.attach(SERVO1_PIN);
  servo2.attach(SERVO2_PIN);
  servo3.attach(SERVO3_PIN);
  
  // Initialize servos to default positions
  servo1.write(0);
  servo2.write(90);
  servo3.write(0);
  
  // Initialize LCD
  Wire.begin();
  lcd.begin(16, 2);
  lcd.setBacklight(255);
  lcd.setCursor(0, 0);
  lcd.print("System Starting");
  delay(2000);
  lcd.clear();
  
  Serial.println("System Ready");
}

void loop() {
  // Reset alerts
  alertActive = false;
  gasAlert = false;
  flameAlert = false;
  motionAlert = false;
  rainAlert = false;
  
  // Clear LCD for new readings
  lcd.clear();
  
  // Read all sensors
  readGasSensor();
  readTemperatureHumidity();
  readFlameSensor();
  readLightSensor();
  readPIRSensor();
  readRainSensor();
  readIRSensor();
  
  // Set overall alert state
  alertActive = gasAlert || flameAlert || motionAlert || rainAlert;
  
  // Handle alert state
  if (alertActive) {
    activateAlert();
  } else {
    deactivateAlert();
  }
  
  delay(2000);
}

void readGasSensor() {
  int gasValue = analogRead(MQ2_PIN);
  Serial.print("Gas Value: ");
  Serial.println(gasValue);
  
  if (gasValue > GAS_THRESHOLD) {
    Serial.println("Gas leak detected!");
    servo1.write(90);
    gasAlert = true;
  } else {
    if (!flameAlert) {  // Only reset if flame isn't controlling it
      servo1.write(0);
    }
  }
}

void readTemperatureHumidity() {
  float temperature = dht.readTemperature();
  float humidity = dht.readHumidity();
  
  if (isnan(temperature) || isnan(humidity)) {
    Serial.println("Failed to read from DHT sensor.");
    lcd.setCursor(0, 0);
    lcd.print("DHT Error!");
  } else {
    Serial.print("Temp: ");
    Serial.print(temperature);
    Serial.println(" C");
    Serial.print("Humidity: ");
    Serial.print(humidity);
    Serial.println(" %");
    
    lcd.setCursor(0, 0);
    lcd.print("T:");
    lcd.print(temperature, 1);
    lcd.print("C H:");
    lcd.print(humidity, 1);
    lcd.print("%");
  }
}

void readFlameSensor() {
  int flameDetected = digitalRead(FLAME_SENSOR_PIN);
  Serial.print("Flame Sensor: ");
  Serial.println(flameDetected);
  
  if (flameDetected == LOW) {  // Assuming LOW means flame detected
    Serial.println("FLAME DETECTED!");
    servo2.write(0);
    servo1.write(0);  // Override gas servo for emergency
    lcd.setCursor(0, 1);
    lcd.print("FLAME!      ");
    flameAlert = true;
  } else {
    servo2.write(90);
    if (!gasAlert) {  // Only reset servo1 if no gas alert
      servo1.write(0);
    }
  }
}

void readLightSensor() {
  int lightValue = analogRead(LDR_PIN);
  Serial.print("Light: ");
  Serial.println(lightValue);
  
  if (lightValue > LIGHT_THRESHOLD) {  // Fixed logic
    Serial.println("Bright light detected!");
    // Add alert if needed
  } else {
    Serial.println("Dim light detected.");
  }
}

void readPIRSensor() {
  int pirState = digitalRead(PIR_PIN);
  
  if (pirState == HIGH) {
    Serial.println("Motion detected!");
    digitalWrite(19, HIGH);
    motionAlert = true;
    lcd.setCursor(0, 1);
    lcd.print("MOTION!     ");
  } else {
    digitalWrite(19, LOW);
  }
}

void readRainSensor() {
  int rainValue = analogRead(RAINDROP_PIN);
  Serial.print("Rain Value: ");
  Serial.println(rainValue);
  
  if (rainValue < RAIN_THRESHOLD) {
    Serial.println("Rain detected!");
    servo3.write(90);
    rainAlert = true;
  } else {
    servo3.write(0);
  }
}

void readIRSensor() {
  int irValue = digitalRead(IR_SENSOR_PIN);
  Serial.print("IR Sensor: ");
  Serial.println(irValue);
  
  if (irValue == LOW) {
    Serial.println("Object detected!");
    digitalWrite(18, HIGH);
  } else {
    digitalWrite(18, LOW);
  }
}

void activateAlert() {
  digitalWrite(LED_RED, HIGH);
  digitalWrite(LED_GREEN, LOW);
  digitalWrite(BUZZER, HIGH);
}

void deactivateAlert() {
  digitalWrite(LED_RED, LOW);
  digitalWrite(LED_GREEN, HIGH);
  digitalWrite(BUZZER, LOW);
}
